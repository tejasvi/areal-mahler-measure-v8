#!/bin/bash
# Pre-freeze checks for L3 on <server>: regenerate every L3 control and compare bytes; reproduce nodes_L3.npy and the certificate.
cd <working directory>
PY=<python>
R=freeze_checks_L3; mkdir -p $R/neg
export OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1 MKL_NUM_THREADS=1 PYTHONDONTWRITEBYTECODE=1
nice -n 10 $PY -I -B make_neg_arl.py cert_arllo_L3.json $R/neg > $R/neg/gen.log 2>&1
nice -n 10 $PY -I -B make_neg2_arl.py cert_arllo_L3.json $R/neg >> $R/neg/gen.log 2>&1
nice -n 10 $PY -I -B make_dip_arl_lo.py cert_arllo_L3.json $R/neg/neg_NARROWNOTCH.json 6129/25000 1/1000000 > $R/neg/notch_gen.log 2>&1
{ for f in $R/neg/*.json; do cmp -s $f ctrl_L3/$(basename $f) && echo "SAME $(basename $f)" || echo "DIFF $(basename $f)"; done
  echo "ctrl_L3 json count: $(ls ctrl_L3/*.json | wc -l), regenerated: $(ls $R/neg/*.json | wc -l)"
  sha256sum $R/neg/neg_NARROWNOTCH.json ctrl_L3/neg_NARROWNOTCH.json
  nice -n 10 $PY -I -B arl_nodes.py A58 primal_L1_arldualN_A58.npy 24 0.01 0.0005 0.004 $R/nodes_L3_repro.npy
  cmp $R/nodes_L3_repro.npy nodes_L3.npy && echo "nodes_L3.npy: BYTE-IDENTICAL"
  nice -n 10 $PY -I -B export_arl_lo.py primal_L3.npy primal_L3_arldualN_A58v.npy A58v 3e-6 $R/cert_arllo_L3_repro.json
  cmp $R/cert_arllo_L3_repro.json cert_arllo_L3.json && echo "cert_arllo_L3.json: BYTE-IDENTICAL"
  echo FREEZECHECKSDONE; } > $R/regen_cmp.txt 2>&1
